// functions/index.js const functions = require('firebase-functions'); const admin = require('firebase-admin'); const nodemailer = require('nodemailer'); admin.initializeApp(); // Configurar transporte de email (usá el que prefieras) const transporter = nodemailer.createTransport({ service: 'gmail', auth: { user: 'tucorreo@gmail.com', pass: 'tucontraseña' } }); exports.enviarReporteEmail = functions.https.onCall(async (data, context) => { try { const { email, comercio, pdfBase64, stats } = data; const mailOptions = { from: 'Hábitat-Car ', to: email, subject: `📊 Reporte de estadísticas - ${comercio}`, html: `

Reporte de estadísticas

Hola ${comercio}, adjuntamos el reporte con tus estadísticas.

Resumen rápido

  • Total canjes: ${stats.total}
  • Canjes LEAD: ${stats.leads}
  • Canjes VIP: ${stats.vips}

El PDF completo está adjunto a este correo.

Este es un mensaje automático, por favor no responder.

`, attachments: [{ filename: `Reporte_${comercio}.pdf`, content: pdfBase64, encoding: 'base64' }] }; await transporter.sendMail(mailOptions); return { success: true, message: 'Email enviado' }; } catch (error) { console.error('Error enviando email:', error); throw new functions.https.HttpsError('internal', error.message); } });
